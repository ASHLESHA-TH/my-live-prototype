import pandas as pd import geopandas as gpd import numpy as np def load_and_clean_wards(geojson_path): """ Step 1: Ingests the 2022 Pune Electoral Ward maps. Cleans geometries and calculates baseline structural area. """ print("Loading PMC Ward Boundaries...") wards = gpd.read_file(geojson_path) # Standardize columns to the mandatory dataset features keep_cols = ['ward_id', 'ward_name', 'geometry'] wards = wards[keep_cols] # Calculate geometric area in square kilometers safely using metric projections if wards.crs.is_geographic: # Project to local UTM Zone (43N for Maharashtra/Pune) to calculate accurate area wards_projected = wards.to_crs(epsg=32643) wards['area_sq_km'] = wards_projected.geometry.area / 1e6 else: wards['area_sq_km'] = wards.geometry.area / 1e6 return wards def merge_demographics(wards_df, census_csv_path): """ Step 2: Merges Census 2011 indicators. Captures raw population segments and high-risk age classes. """ print("Merging Census demographic variables...") census = pd.read_csv(census_csv_path) # Expected inputs: ward_id, total_pop, child_pop_0_6, elderly_pop_60_plus # Perform a left join to preserve geospatial integrity merged = wards_df.merge(census, on="ward_id", how="left") # Calculate derived structural variables merged['population_density'] = merged['total_pop'] / merged['area_sq_km'] merged['children_pct'] = (merged['child_pop_0_6'] / merged['total_pop']) * 100 merged['elderly_pct'] = (merged['elderly_pop_60_plus'] / merged['total_pop']) * 100 return merged def merge_slum_data(wards_df, slums_csv_path): """ Step 3: Appends PMC informal settlement metrics. Integrates density characteristics and housing distributions. Aggregates large micro-data coordinates into localized density metrics. """ print("Aggregating large scale tree census distributions...") # Loading individual trees can hit memory bottlenecks; chunking or optimizing is recommended trees = pd.read_csv(raw_trees_csv_path, usecols=['ward_id', 'canopy_dia_m', 'condition']) # Compute structural groupby summary configurations tree_summary = trees.groupby('ward_id').agg( tree_count=('ward_id', 'count'), avg_canopy=('canopy_dia_m', 'mean'), healthy_count=('condition', lambda x: (x == 'Healthy').sum()) ).reset_index() # Merge structural aggregates back to main master tracking pipeline merged = wards_df.merge(tree_summary, on="ward_id", how="left") merged['tree_count'] = merged['tree_count'].fillna(0) merged['tree_density'] = merged['tree_count'] / merged['area_sq_km'] merged['healthy_tree_percentage'] = (merged['healthy_count'] / merged['tree_count'].replace(0, np.nan)) * 100 merged['healthy_tree_percentage'] = merged['healthy_tree_percentage'].fillna(0) merged.drop(columns=['healthy_count'], inplace=True) return merged def simulate_satellite_zonal_stats(wards_df): """ Step 5: Placeholder for Google Earth Engine / GIS spatial intersections.
